cmake_minimum_required(VERSION 3.16)

# The Vita toolchain file must be selected before project().
# build.sh sets VITASDK; CMake can also be invoked manually with
# -DCMAKE_TOOLCHAIN_FILE=$VITASDK/share/vita.toolchain.cmake.
if(NOT DEFINED CMAKE_TOOLCHAIN_FILE)
  if(DEFINED ENV{VITASDK})
    set(CMAKE_TOOLCHAIN_FILE "$ENV{VITASDK}/share/vita.toolchain.cmake"
        CACHE PATH "Vita toolchain file")
  else()
    message(FATAL_ERROR "VITASDK is not set. Run ./build.sh or export VITASDK.")
  endif()
endif()

project(cockpit C)

# Provides vita_create_self() and vita_create_vpk().
include("${VITASDK}/share/vita.cmake" REQUIRED)

set(VITA_APP_NAME "COCKPIT")
set(VITA_TITLEID  "COCKPIT01")  # Exactly 9 uppercase characters.
set(VITA_VERSION  "01.00")      # Must match ##.##.

# Monotonic build number, supplied by build.sh. It goes in the artifact name so
# successive builds are distinguishable on the console and in ux0:data; it is
# deliberately NOT part of APP_VER, which must stay ##.##.
set(COCKPIT_BUILD "0" CACHE STRING "Build number, set by build.sh")

# UNSAFE is the default (D42, superseding D9). Reading `uma0:` capacity needs
# sceIoDevctl, which the safe authid refuses; there is no safe route to a
# non-ux0: volume. --safe builds the restricted variant and is marked in the
# artifact name.
option(COCKPIT_SAFE "Build a safe eboot (no ux0:/ur0: access)" OFF)

set(COCKPIT_VARIANT "")
if(COCKPIT_SAFE)
  set(COCKPIT_VARIANT "-safe")
endif()

set(COCKPIT_VPK "cockpit-v${VITA_VERSION}-b${COCKPIT_BUILD}${COCKPIT_VARIANT}.vpk")

# The 2026.08 host tool leaves VERSION/CONTENT_ID empty unless told otherwise.
set(VITA_MKSFOEX_FLAGS "${VITA_MKSFOEX_FLAGS} -s VERSION=${VITA_VERSION}")
set(VITA_MKSFOEX_FLAGS "${VITA_MKSFOEX_FLAGS} -s CONTENT_ID=HB0001-${VITA_TITLEID}_00-0000000000000000")

set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -Wall -Wextra")

# Reserve 64 KiB of headroom before the RW segment. Without this, some
# code sizes make vita-elf-create fail with "segment 1 overlaps".
set(CMAKE_EXE_LINKER_FLAGS "${CMAKE_EXE_LINKER_FLAGS} -Wl,--defsym=__sce_headroom=0x10000")

add_executable(cockpit
  src/main.c
  src/app/app.c
  src/app/screen.c
  src/core/history.c
  src/core/input.c
  src/core/metrics.c
  src/core/plugin_editor.c
  src/core/plugins.c
  src/gfx/draw.c
  src/gfx/font.c
  src/gfx/theme.c
  src/panels/core.c
  src/panels/frame.c
  src/panels/plugins.c
  src/panels/power.c
  src/panels/registry.c
  src/panels/storage.c
  src/panels/system.c
  src/platform/platform_psp2.c
  src/ui/header.c
  src/ui/modal.c
  src/ui/tabbar.c
)

target_include_directories(cockpit PRIVATE src)

# Only the libraries the source calls, plus libvita2d's own hard requirements.
# SceAppMgr and SceCommonDialog are not called from this source; libvita2d's
# init path references sceAppMgrGetBudgetInfo and the shared-framebuffer calls,
# so the link fails without them. The *_stub libraries are NID import stubs
# generated by VitaSDK, not normal shared libraries.
target_link_libraries(cockpit
  vita2d
  SceDisplay_stub
  SceGxm_stub
  SceSysmodule_stub
  SceCtrl_stub
  SceAppMgr_stub
  SceCommonDialog_stub
  SceLibKernel_stub
  ScePower_stub
  SceSysmem_stub
  SceIofilemgr_stub
  SceKernelModulemgr_stub
  SceRtc_stub
  m
)

# Authid follows COCKPIT_SAFE: ...0002 safe, ...0001 unsafe (D42).
if(COCKPIT_SAFE)
  vita_create_self(cockpit.self cockpit)
else()
  vita_create_self(cockpit.self cockpit UNSAFE)
endif()

vita_create_vpk(${COCKPIT_VPK} ${VITA_TITLEID} cockpit.self
  VERSION ${VITA_VERSION}
  NAME ${VITA_APP_NAME}
  FILE sce_sys/icon0.png sce_sys/icon0.png
  FILE sce_sys/livearea/contents/bg.png sce_sys/livearea/contents/bg.png
  FILE sce_sys/livearea/contents/startup.png sce_sys/livearea/contents/startup.png
  FILE sce_sys/livearea/contents/template.xml sce_sys/livearea/contents/template.xml
)
